        let soundEnabled = true;

        function getAudioContext() {
            if (!audioCtx) {
                const AudioContextClass = window.AudioContext || window.webkitAudioContext;
                if (AudioContextClass) {
                    audioCtx = new AudioContextClass();
                }
            }
            if (audioCtx && audioCtx.state === 'suspended') {
                audioCtx.resume().catch(() => {});
            }
            return audioCtx;
        }

        function playTone(freq, duration, delay = 0, type = 'sine') {
            try {
                const ctx = getAudioContext();
                if (!ctx) return;
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = type;
                osc.frequency.setValueAtTime(freq, ctx.currentTime + delay);
                
                gain.gain.setValueAtTime(0.15, ctx.currentTime + delay);
                gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + delay + duration);
                
                osc.connect(gain);
                gain.connect(ctx.destination);
                
                osc.start(ctx.currentTime + delay);
                osc.stop(ctx.currentTime + delay + duration);
            } catch (e) {
                // Abaikan jika perangkat dalam mode senyap
            }
        }

                            <span class="text-[10px] font-extrabold text-amber-700 uppercase block mb-0.5">${labels[i]}</span>
                            <span class="text-xl text-amber-300 font-bold">?</span>
                        </div>
                    `;
                }
            }

            choicesContainer.innerHTML = '';
            s2ShuffledChoices.forEach(item => {
                const isAlreadySelected = s2SelectedOrder.some(sel => sel.id === item.id);
                if (isAlreadySelected) {
                    choicesContainer.innerHTML += `
                        <div class="bg-slate-100 border-2 border-slate-200 rounded-2xl p-3 text-center opacity-40">
                            <div class="text-3xl sm:text-4xl mb-1">${item.emoji}</div>
                            <div class="font-bold text-slate-500 text-xs sm:text-sm">${item.name}</div>
                            <span class="text-[10px] font-bold text-emerald-600">Terpilih ✓</span>
                        </div>
                    `;
                } else {
                    choicesContainer.innerHTML += `
                        <button type="button" onclick="selectStage2Item(${item.id})" class="btn-3d bg-white active:bg-amber-50 border-2 border-amber-300 rounded-2xl p-3 sm:p-4 text-center">
                            <div class="text-3xl sm:text-5xl mb-1.5">${item.emoji}</div>
                            <div class="font-title font-bold text-slate-800 text-xs sm:text-base">${item.name}</div>
                        </button>
                    `;
                }
            });
        }

        function selectStage2Item(itemId) {
            const quest = stage2Data[s2Index];
            const item = quest.items.find(i => i.id === itemId);
            if (!item || s2SelectedOrder.length >= 4) return;

            playSound('click');
            s2SelectedOrder.push(item);
            renderStage2Board();

            if (s2SelectedOrder.length === 4) {
                setTimeout(() => {
                    const isOrderCorrect = s2SelectedOrder.every((val, idx) => val.id === idx + 1);
                    if (isOrderCorrect) {
                        playSound('correct');
                        correctCount++;
                        addScore(15);
                        showFeedback(true, "Rantai Makanan Sempurna! 🔗", quest.explanation);
                    } else {
                        playSound('wrong');
                        showFeedback(false, "Urutannya Belum Tepat! 🤔", quest.explanation);
                    }
                }, 300);
            }
        }

        function resetStage2Selection() {
            playSound('click');
            s2SelectedOrder = [];
            renderStage2Board();
        }

        // --- 8. LOGIKA BABAK 3 (PAHLAWAN PENJAGA ALAM) ---
        function startStage3() {
            currentStage = 3;
            showScreen('screen-stage3');
            updateHeaderProgress(3);
            renderStage3Question();
        }

        function renderStage3Question() {
            const quest = stage3Data[s3Index];
            document.getElementById('s3-progress').textContent = `${s3Index + 1} / ${stage3Data.length}`;
            document.getElementById('s3-badge').textContent = quest.badge;
            document.getElementById('s3-icon').textContent = quest.icon;
            document.getElementById('s3-question').textContent = quest.question;

            const optionsContainer = document.getElementById('s3-options');
            optionsContainer.innerHTML = '';
            const letters = ['A', 'B', 'C'];

            quest.options.forEach((opt, idx) => {
                optionsContainer.innerHTML += `
                    <button type="button" onclick="checkStage3(${idx})" class="btn-3d w-full bg-white active:bg-emerald-50 border-2 border-emerald-300 rounded-2xl p-3.5 sm:p-4 text-left flex items-center gap-3">
                        <span class="w-9 h-9 sm:w-10 sm:h-10 rounded-xl bg-emerald-100 text-emerald-800 font-title font-bold text-base sm:text-lg flex items-center justify-center shrink-0 border border-emerald-300">
                            ${letters[idx]}
                        </span>
                        <span class="font-bold text-slate-700 text-xs sm:text-base leading-snug">${opt.text}</span>
                    </button>
                `;
            });
        }

        function checkStage3(optionIndex) {
            const quest = stage3Data[s3Index];
            const chosen = quest.options[optionIndex];

            if (chosen.correct) {
                playSound('correct');
                correctCount++;
                addScore(15);
                showFeedback(true, "Pahlawan Alam Sejati! 🌱", quest.explanation);
            } else {
                playSound('wrong');
                showFeedback(false, "Kurang Tepat, Sahabat Alam! 🌿", quest.explanation);
            }
        }

        // --- 9. SISTEM MODAL UMPAN BALIK & TOMBOL LANJUT ---
        function showFeedback(isCorrect, title, message) {
            const modal = document.getElementById('feedback-modal');
            const box = document.getElementById('feedback-box');
            const emojiEl = document.getElementById('fb-emoji');
            const titleEl = document.getElementById('fb-title');
            const msgEl = document.getElementById('fb-message');
            const btnEl = document.getElementById('fb-next-btn');

            if (isCorrect) {
                box.className = "bg-white rounded-3xl max-w-md w-full p-5 sm:p-7 text-center shadow-2xl border-4 border-emerald-400 animate-pop";
                emojiEl.textContent = "🎉";
                titleEl.className = "font-title text-xl sm:text-3xl font-bold text-emerald-700 mb-2";
                btnEl.className = "btn-3d w-full bg-emerald-500 text-white font-title text-lg sm:text-xl font-bold py-3.5 rounded-2xl flex items-center justify-center gap-2";
            } else {
                box.className = "bg-white rounded-3xl max-w-md w-full p-5 sm:p-7 text-center shadow-2xl border-4 border-amber-400 animate-pop";
                emojiEl.textContent = "💡";
                titleEl.className = "font-title text-xl sm:text-3xl font-bold text-amber-700 mb-2";
                btnEl.className = "btn-3d w-full bg-amber-500 text-white font-title text-lg sm:text-xl font-bold py-3.5 rounded-2xl flex items-center justify-center gap-2";
            }

            titleEl.textContent = title;
            msgEl.textContent = message;
            modal.classList.remove('hidden');
        }

        function handleNextStep() {
            playSound('click');
            document.getElementById('feedback-modal').classList.add('hidden');

            if (currentStage === 1) {
                s1Index++;
                if (s1Index < stage1Data.length) {
                    renderStage1Question();
                } else {
                    startStage2();
                }
            } else if (currentStage === 2) {
                s2Index++;
                if (s2Index < stage2Data.length) {
                    renderStage2Question();
                } else {
                    startStage3();
                }
            } else if (currentStage === 3) {
                s3Index++;
                if (s3Index < stage3Data.length) {
                    renderStage3Question();
                } else {
                    finishGame();
                }
            }
        }

        // --- 10. HALAMAN PENUTUP & PERHITUNGAN NILAI AKHIR ---
        function finishGame() {
            currentStage = 4;
            playSound('win');
            showScreen('screen-end');

            const finalGrade = Math.round((correctCount / totalQuestions) * 100);
            document.getElementById('final-score').textContent = finalGrade;
            document.getElementById('final-stats').textContent = `Jawaban Benar: ${correctCount} dari ${totalQuestions} Tantangan (Skor Poin: ${score})`;

            let stars = "⭐⭐⭐";
            let predicate = "Predikat: Khalifah Penjaga Alam Terbaik! 🌟";

            if (finalGrade >= 85) {
                stars = "⭐⭐⭐";
                predicate = "Predikat: Khalifah Penjaga Alam Terbaik! 🌟";
            } else if (finalGrade >= 65) {
                stars = "⭐⭐";
                predicate = "Predikat: Penjelajah Ekosistem Hebat! 🌿";
            } else {
                stars = "⭐";
                predicate = "Predikat: Sahabat Alam Semangat Belajar! 🌱";
            }

            document.getElementById('final-stars').textContent = stars;
            document.getElementById('final-predicate').textContent = predicate;
        }
